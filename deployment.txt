# AI-Coding-Share 部署文档
# 服务器: Lighthouse HK (150.109.75.152)
# 系统: Ubuntu 24.04 LTS
# 用户: ubuntu

========================================
一、首次部署步骤
========================================

1. 拉取项目
---
cd /home/ubuntu
git clone git@github.com:YicongCao/AI-Coding-Share.git
cd AI-Coding-Share

2. 构建 Docker 镜像
---
sudo docker build -t ai-coding-share:latest .

3. 启动容器 (绑定 localhost，不暴露公网)
---
sudo docker run -d -p 127.0.0.1:5174:5174 --name ai-coding-share-run ai-coding-share:latest

4. 安装 Nginx
---
sudo apt update && sudo apt install -y nginx

5. 配置 Nginx 反向代理
---
sudo tee /etc/nginx/sites-available/ai-show.yicong.me > /dev/null << NGINX
server {
    listen 80;
    server_name ai-show.yicong.me;

    location / {
        proxy_pass http://127.0.0.1:5174;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400;
    }
}
NGINX

sudo ln -sf /etc/nginx/sites-available/ai-show.yicong.me /etc/nginx/sites-enabled/
sudo nginx -t && sudo systemctl reload nginx

6. 配置 Cloudflare DNS 及 HTTPS
---
a) 登录 Cloudflare Dashboard，进入 yicong.me 域名管理页面
b) DNS -> Records -> Add Record:
   - Type: A
   - Name: ai-show
   - Content: 150.109.75.152
   - Proxy status: Proxied (橙色云朵，开启 Cloudflare 代理)
c) SSL/TLS -> Overview:
   - 加密模式选择 "Full"（因为源站 Nginx 监听 80，Cloudflare 负责终止 HTTPS）
   - 如果源站后续配了自签证书或 Cloudflare Origin Certificate，可切换为 "Full (Strict)"
d) SSL/TLS -> Edge Certificates:
   - 确认 "Always Use HTTPS" 已开启
   - 可选开启 "Automatic HTTPS Rewrites"

配置完成后，用户通过 https://ai-show.yicong.me/ 访问时:
  用户 --HTTPS--> Cloudflare CDN --HTTP(80)--> 源站 Nginx --> localhost:5174

7. (可选) Docker 容器开机自启
---
sudo docker update --restart unless-stopped ai-coding-share-run

配置后服务器重启时 Docker 会自动拉起容器，无需手动操作。

========================================
二、服务器重启后恢复服务
========================================

# 如果已配置 --restart unless-stopped，容器会自动启动，无需手动操作。
# 如果未配置，需手动启动:
sudo docker start ai-coding-share-run

# Nginx 已设为开机自启，无需手动操作。

# 确认服务状态:
sudo systemctl status nginx
sudo docker ps

# 如果容器不存在 (被删除过)，需要重新创建:
cd /home/ubuntu/AI-Coding-Share
sudo docker build -t ai-coding-share:latest .
sudo docker run -d -p 127.0.0.1:5174:5174 --restart unless-stopped --name ai-coding-share-run ai-coding-share:latest

========================================
三、日常运维命令
========================================

# 查看容器日志
sudo docker logs -f ai-coding-share-run

# 重启容器
sudo docker restart ai-coding-share-run

# 更新代码并重新部署
cd /home/ubuntu/AI-Coding-Share
git pull
sudo docker stop ai-coding-share-run && sudo docker rm ai-coding-share-run
sudo docker build -t ai-coding-share:latest .
sudo docker run -d -p 127.0.0.1:5174:5174 --restart unless-stopped --name ai-coding-share-run ai-coding-share:latest

# 查看磁盘使用情况
df -h /
